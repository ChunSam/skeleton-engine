빠른 시작

use engine::*;

struct MySystem;
impl System for MySystem {
    fn run(&mut self, world: &mut World, _dt: f32) {
        let should_quit = world
            .resource::<InputState>()
            .map(|input| input.just_pressed(KeyCode::Escape))
            .unwrap_or(false);

        if should_quit {
            world.resource_mut::<ShouldQuit>().unwrap().quit();
        }
    }
}

fn main() {
    let mut app = App::new();

    // 창 설정
    app.world.insert_resource(WindowConfig {
        width: 1280,
        height: 720,
        title: "My Game".to_string(),
        clear_color: [0.05, 0.05, 0.1, 1.0],
    });

    // 이미지 로드 (핫 리로드 지원 Handle 반환)
    app.load_image("assets/player.png");

    // 엔티티 스폰
    let e = app.world.spawn();
    app.world.add_component(e, Transform {
        position: Vec2::new(400.0, 300.0),
        scale: Vec2::splat(64.0),
        rotation: 0.0,
        z: 0.0,
    });
    app.world.add_component(e, Sprite::textured("assets/player.png"));

    app.add_system(MySystem);
    app.run();
}

App

App은 엔진의 진입점. 시스템·씬·리소스를 조립하고 run()으로 게임 루프를 시작한다.

let mut app = App::new();
메서드설명
app.add_system(s)시스템 등록. 매 프레임 등록 순서대로 실행됨
app.load_image("path.png")PNG 이미지 로드 — Handle<ImageAsset> 반환, 핫 리로드 참여 (run() 전·후 모두 호출 가능)
app.register_event::<E>()이벤트 타입 등록 (자동 flush 포함)
app.set_scene(Box<dyn Scene>)씬을 즉시 전환 (run() 전·후 모두 가능)
app.run()이벤트 루프 시작 (블로킹)

프레임 실행 순서:

  1. 시스템들 순서대로 run(dt) 호출
  2. 이벤트 큐 flush
  3. 입력 상태 flush
  4. 씬 전환 명령 처리
  5. 렌더링 (scene -> post(optional) -> lighting(optional) -> final)

쿼리

메서드반환설명
world.query::<A>()(Entity, &A)A를 가진 모든 엔티티
world.query2::<A, B>()(Entity, &A, &B)A와 B를 모두 가진 엔티티
world.query3::<A, B, C>()(Entity, &A, &B, &C)A·B·C 모두
world.query4::<A, B, C, D>()(Entity, &A, &B, &C, &D)A·B·C·D 모두
world.query_opt2::<A, B>()(Entity, &A, Option<&B>)A 필수, B는 있으면 Some
world.query_mut::<A>()(Entity, &mut A)A를 가변으로 순회
world.query2_mut::<A, B>()(Entity, &mut A, &mut B)A·B를 동시에 가변 순회 (서로 다른 타입)
world.query3_mut::<A, B, C>()(Entity, &mut A, &mut B, &mut C)A·B·C 동시 가변 (서로 다른 타입)

쿼리는 Iterator를 반환하므로 .for_each(), .collect(), .filter() 등을 그대로 사용한다.

주의: 같은 World에서 불변 쿼리 중 get_mut을 섞으면 borrow 충돌이 발생한다.
패턴: 먼저 .collect()로 엔티티 목록을 뽑은 뒤 순회하며 get_mut을 호출한다.
// 올바른 패턴
let targets: Vec<Entity> = world.query::<Enemy>().map(|(e, _)| e).collect();
for e in targets {
    if let Some(hp) = world.get_mut::<Health>(e) {
        hp.value -= 10;
    }
}
가변 쿼리(query_mut/query2_mut/query3_mut): 단일·다중 컴포넌트를 제자리에서 변경할 때 위의 collect → get_mut 우회 없이 바로 순회한다. 다중 컴포넌트 버전은 HashMap::get_disjoint_mut으로 서로 다른 타입을 한 번에 빌려오므로(타입이 모두 달라야 함, 같으면 패닉) borrow 충돌이 없다.
// 위치 + 속도를 한 번에 가변 순회 (collect 불필요)
// Velocity는 게임이 정의한 예시 컴포넌트 (struct Velocity(Vec2))
for (e, t, vel) in world.query2_mut::<Transform, Velocity>() {
    t.position += vel.0 * dt;
}

변경 감지 주의: 가변 쿼리로 바꾼 값은 query_changed::<T>()에 자동 기록되지 않는다. 변경 감지가 필요하면 world.mark_changed::<T>(entity)를 호출한다.

쿼리 필터 (With / Without)

// Sprite가 있는 Transform만
for (e, t) in world.query_with::<Transform, Sprite>() {
    // t: &Transform
}

// Enemy 컴포넌트가 없는 Transform만
for (e, t) in world.query_without::<Transform, Enemy>() {
    // t: &Transform
}

씬 전환

앱 초기화 / 외부 제어 시 (App 편의 메서드):

app.set_scene(Box::new(MenuScene));    // Replace — 월드 리셋 후 새 씬
app.push_scene(Box::new(PauseMenu));   // Push   — 현재 씬 위에 오버레이 (아래 씬 보존)
app.pop_scene();                       // Pop    — 최상위 씬 제거, 아래 씬 재개 (빈 스택이면 no-op)

set_scene/push_scene/pop_scene은 각각 SceneCmd::Replace/Push/Pop을 즉시 적용하는 App 레벨 단축 메서드다 (시스템 바깥, 예: run() 호출 전이나 메인 루프를 직접 모는 코드에서). 시스템 안에서는 아래의 SceneChange 리소스를 쓴다.

시스템 내부에서 (런타임):

fn run(&mut self, world: &mut World, _dt: f32) {
    if let Some(sc) = world.resource_mut::<SceneChange>() {
        // 전체 교체 (월드 리셋 포함)
        sc.request(SceneCmd::Replace(Box::new(GamePlay)));

        // 현재 씬 위에 쌓기 (일시정지 메뉴 등)
        sc.request(SceneCmd::Push(Box::new(PauseMenu)));

        // 최상위 씬 제거
        sc.request(SceneCmd::Pop);
    }
}
명령동작
Replace(scene)스택 전부 종료 → 월드 리셋 → 새 씬 진입
Push(scene)현재 월드 유지, 새 씬의 on_enter만 호출
Pop최상위 씬 종료, 해당 씬이 등록한 시스템 제거

같은 프레임에 여러 번 request하면 마지막 명령만 실행된다.

입력

InputState 리소스를 통해 키보드·마우스 입력을 읽는다.

use winit::keyboard::KeyCode;
use winit::event::MouseButton;

fn run(&mut self, world: &mut World, _dt: f32) {
    let input = world.resource::<InputState>().unwrap();

    // 키보드
    input.just_pressed(KeyCode::Space)     // 이번 프레임에 눌림 (1프레임만 true)
    input.is_pressed(KeyCode::ArrowLeft)   // 현재 누르고 있는 중
    input.just_released(KeyCode::Space)    // 이번 프레임에 뗌

    // 마우스
    input.cursor()                                   // Vec2 (스크린 픽셀 좌표)
    input.mouse_just_pressed(MouseButton::Left)      // 이번 프레임에 클릭
    input.is_mouse_pressed(MouseButton::Left)        // 현재 누르고 있는 중
    input.mouse_just_released(MouseButton::Left)     // 이번 프레임에 뗌
    input.scroll()                                   // f32, 프레임마다 초기화

    // 문자 입력 버퍼 (TextInput과 함께 사용)
    input.text_chars()   // &[char], '\x08'=Backspace '\n'=Enter, 프레임마다 초기화
}
커서 좌표는 스크린 픽셀 기준(좌상단 원점). 월드 좌표로 변환하려면 Camera::screen_to_world()를 사용한다.

InputMap (키 리바인딩)

게임패드

GamepadState 리소스로 최대 4개 패드(슬롯 0–3)를 읽는다. App::new()에서 자동 삽입되며 네이티브 전용(gilrs)이다 — wasm에서는 항상 미연결로 동작한다.

use engine::{GamepadButton, GamepadAxis};

if let Some(gs) = world.resource::<GamepadState>() {
    let pad = gs.primary().unwrap_or(0);            // 첫 연결 패드 슬롯
    gs.is_pressed(pad, GamepadButton::South);       // 누르고 있는 중 (A / Cross)
    gs.just_pressed(pad, GamepadButton::Start);     // 이번 프레임에 눌림
    let lx = gs.axis(pad, GamepadAxis::LeftStickX); // -1.0 ~ 1.0 (데드존 미적용)
    gs.any_connected();                             // 패드 하나라도 연결됨?
}
macOS 주의: gilrs는 IOKit HID로 패드를 읽지만, 최신 macOS는 Xbox·PlayStation 패드를 GameController 프레임워크로 배타 점유한다. 그런 패드는 연결은 감지돼도(Connected 이벤트) 축·버튼 입력이 gilrs로 전달되지 않는다(무반응). Linux·Windows에서는 정상 동작하며, macOS에서는 Apple 드라이버에 점유되지 않는 일반 HID(DInput) 패드만 읽힌다. (네이티브 게임패드는 native-only — wasm에서는 항상 미연결.)

InputMap 게임패드 바인딩

하나의 액션에 키보드 키·게임패드 버튼·게임패드 축을 함께 묶을 수 있다(가산적 — 바인딩 중 하나라도 활성이면 액션 활성). AxisBinding은 아날로그 축을 임계값 기준의 디지털 액션으로 바꾼다.

use engine::{InputMap, GamepadButton, GamepadAxis, AxisBinding};
use winit::keyboard::KeyCode;

let mut map = InputMap::new();
map.bind("jump", KeyCode::Space);                      // 키보드
map.bind_gamepad_button("jump", GamepadButton::South); // + 게임패드 버튼
// 왼쪽 스틱을 왼쪽으로(값 <= -0.5) 밀면 "move_left" 활성
map.bind_gamepad_axis("move_left", AxisBinding::negative(GamepadAxis::LeftStickX, 0.5));
map.bind_gamepad_axis("move_right", AxisBinding::positive(GamepadAxis::LeftStickX, 0.5));

// 시스템에서: 키보드 OR 게임패드를 함께 평가 (pad = 슬롯 인덱스)
let input = world.resource::<InputState>().unwrap();
let gs = world.resource::<GamepadState>().unwrap();
if map.is_pressed_with_gamepad("move_left", input, gs, 0) { ... }
if map.just_pressed_with_gamepad("jump", input, gs, 0) { ... }
메서드설명
bind_gamepad_button(action, button)액션에 게임패드 버튼 추가 (가산적)
bind_gamepad_axis(action, AxisBinding)액션에 축 + 방향 + 임계값 추가
AxisBinding::positive(axis, threshold)값 >= threshold일 때 활성
AxisBinding::negative(axis, threshold)값 <= -threshold일 때 활성
is_pressed_with_gamepad(a, input, gs, pad)키보드/버튼/축 중 하나라도 활성
just_pressed_with_gamepad(a, input, gs, pad)이번 프레임 활성화 (축은 "현재 활성"으로만 평가 — 아래 참고)
just_released_with_gamepad(a, input, gs, pad)이번 프레임 해제 (키 / 버튼만)
keys_for(action)액션에 묶인 모든 키보드 키 &[KeyCode]
GamepadState는 디지털 버튼만 just_pressed/just_released 엣지를 추적한다. 따라서 just_pressed_with_gamepad의 축 바인딩은 "현재 임계값을 넘었는지"로만 평가되며, 축 엣지 검출이 필요하면 호출 측에서 쿨다운을 둔다. 키보드 전용 메서드(is_pressed/just_pressed/just_released)는 이전과 동일하게 동작한다.

렌더링

Transform + Sprite

모든 렌더링 가능한 오브젝트는 Transform과 Sprite 컴포넌트를 가진다.

// 텍스처 스프라이트
world.add_component(e, Transform {
    position: Vec2::new(100.0, 200.0),
    scale: Vec2::new(64.0, 64.0),  // 픽셀 크기
    rotation: 0.0,                  // 라디안 (Z축)
    z: 0.0,                         // 깊이 (클수록 앞에 그려짐)
});
world.add_component(e, Sprite::textured("assets/hero.png"));

// 단색 사각형
world.add_component(e, Sprite::colored(1.0, 0.0, 0.0));  // RGB

// 핸들이 있으면 우선 사용하고, 없으면 경로 fallback
world.add_component(e, Sprite::textured_with_handle("assets/hero.png", maybe_handle));

// 색상 배율 오버라이드
world.add_component(e, Sprite {
    texture: Some("assets/hero.png".into()),
    color: [1.0, 0.5, 0.5, 0.8],  // RGBA 배율 (텍스처와 곱해짐)
    ..Default::default()
});

z 값이 클수록 앞에 그려진다. 같은 z값이면 스폰 순서에 따름.

텍스트 렌더링

TextQueue 리소스에 DrawText를 넣으면 다음 프레임에 렌더링된다.

use engine::{DrawText, TextQueue};

fn run(&mut self, world: &mut World, _dt: f32) {
    if let Some(tq) = world.resource_mut::<TextQueue>() {
        tq.push(DrawText {
            text: format!("HP: {}", self.hp),
            position: Vec2::new(10.0, 10.0),  // 스크린 좌표
            size: 24.0,                         // 폰트 크기 (픽셀)
            color: [1.0, 1.0, 1.0, 1.0],       // RGBA
        });
    }
}
텍스트는 스크린 스페이스에 그려진다 (카메라 영향 없음).

UI 사각형

UiQueue에 DrawRect를 넣으면 스프라이트 위, 텍스트 아래의 screen-space UI primitive 레이어에 그려진다. DrawRect와 DrawImage는 같은 z 기준으로 정렬되며, 같은 z에서는 이미지가 먼저, 사각형이 나중에 그려진다.

use engine::{DrawRect, UiQueue};

let rect = DrawRect::new(x, y, width, height, [r, g, b, a]).with_z(0.9);
world.resource_mut::<UiQueue>().unwrap().push(rect);

UI 이미지

UiImageQueue에 DrawImage를 넣으면 카메라 영향을 받지 않는 screen-space 텍스처 쿼드를 같은 UI primitive 레이어에 그린다.

use engine::{DrawImage, UiImageQueue, UvRect};

let icon = DrawImage::textured(16.0, 16.0, 32.0, 32.0, "assets/icons.png")
    .with_uv(UvRect::from_grid(2, 0, 8, 4))
    .with_z(0.95);
world.resource_mut::<UiImageQueue>().unwrap().push(icon);

DrawImage::textured_with_handle(...)는 Handle<ImageAsset>가 있으면 우선 사용하고, 없으면 경로를 fallback으로 보존한다.

애니메이션

스프라이트시트 애니메이션은 AnimationPlayer 컴포넌트로 처리한다.

use engine::{AnimationClip, AnimationPlayer, UvRect};

// 4열 2행 스프라이트시트, 첫 번째 행 = 달리기 (4프레임)
let run_clip = AnimationClip {
    frames: (0..4).map(|col| UvRect::from_grid(col, 0, 4, 2)).collect(),
    fps: 12.0,
    looping: true,
};

// 두 번째 행 = 점프 (4프레임)
let jump_clip = AnimationClip {
    frames: (0..4).map(|col| UvRect::from_grid(col, 1, 4, 2)).collect(),
    fps: 8.0,
    looping: false,
};

let player = AnimationPlayer::new(vec![run_clip, jump_clip]);
world.add_component(e, player);
world.add_component(e, Sprite::textured("assets/hero_sheet.png"));

픽셀 단위 crop은 UvRect::from_pixels(...)로 표현한다. 의도적인 세로 미러링이 필요할 때만 .flipped_y()를 추가한다.

let uv = UvRect::from_pixels(44.0, 136.0, 151.0, 164.0, 1254.0, 1254.0);
world.add_component(e, uv);

시스템에서 클립 전환:

if let Some(anim) = world.get_mut::<AnimationPlayer>(entity) {
    anim.play(1);  // 클립 인덱스 0 = 달리기, 1 = 점프
}

AnimationSystem을 등록해야 프레임이 자동으로 진행된다:

app.add_system(AnimationSystem::new());

크로스페이드 전환

play_with_crossfade(clip_index, duration)으로 지정한 시간(초)에 걸쳐 부드럽게 전환한다. 전환 중 BlendWeight 컴포넌트가 자동으로 갱신되며, 게임 코드에서 알파 보간 등에 활용할 수 있다.

use engine::{AnimationPlayer, BlendWeight};

// 0.2초 크로스페이드
if let Some(player) = world.get_mut::<AnimationPlayer>(entity) {
    player.play_with_crossfade(1, 0.2);
}

// 전환 진행도 읽기 (0.0 = 이전 클립, 1.0 = 새 클립 / 전환 없으면 1.0)
if let Some(bw) = world.get_mut::<BlendWeight>(entity) {
    // bw.0 : f32 [0.0 ~ 1.0]
}

1D 블렌드 트리

BlendTree1D는 float 파라미터에 따라 자동으로 클립을 선택하고 크로스페이드 전환을 수행한다. 이동 속도 → 애니메이션 자동 전환 같은 패턴에 유용하다.

use engine::{BlendEntry, BlendTree1D, BlendTreeSystem};

// 트리 구성 (threshold 오름차순)
let tree = BlendTree1D::new(
    vec![
        BlendEntry { threshold: 0.0, clip_index: 0 },  // idle
        BlendEntry { threshold: 0.3, clip_index: 1 },  // walk
        BlendEntry { threshold: 1.2, clip_index: 2 },  // run
    ],
    0.15,  // 클립 전환 시 크로스페이드 지속 시간 (초)
);
world.add_component(entity, tree);

// 시스템 등록 (BlendTreeSystem은 AnimationSystem 이전에)
app.add_system(BlendTreeSystem::new());
app.add_system(AnimationSystem::new());

// 매 프레임 파라미터 갱신
if let Some(tree) = world.get_mut::<BlendTree1D>(entity) {
    tree.set_param(speed);  // speed에 따라 idle / walk / run 자동 전환
}

상태 머신 전환 크로스페이드

AnimationStateMachine의 전환에 크로스페이드 시간을 지정할 수 있다. add_transition은 기존처럼 즉시 전환(하드 스위치)이고, add_transition_crossfade는 전환이 발동될 때 지정한 시간(초)에 걸쳐 play_with_crossfade 경로(2-UV 셰이더 블렌드, BlendTreeSystem과 동일 메커니즘)로 부드럽게 블렌드한다.

use engine::{AnimationStateMachine, StateMachineSystem, TransitionCond};

let mut sm = AnimationStateMachine::new("idle", 0);
sm.add_state("walk", 1);

// 즉시 전환 (기존 동작과 동일)
sm.add_transition("idle", "walk", vec![TransitionCond::FloatGt("speed".into(), 0.1)]);

// 0.2초 크로스페이드 전환
sm.add_transition_crossfade(
    "walk", "idle",
    vec![TransitionCond::FloatLt("speed".into(), 0.1)],
    0.2,
);

하드 스위치와 크로스페이드를 나란히 비교하는 예제: cargo run --example sm_crossfade (사전에 gen_blend_sheet 실행 필요).

물리 (Rapier2D)

시뮬레이션 진행 및 Transform 동기화

PhysicsSystem은 매 프레임 PhysicsWorld를 스텝하고 Transform을 갱신한다.

impl System for MyPhysicsSystem {
    fn run(&mut self, world: &mut World, dt: f32) {
        self.physics.step(dt);

        let bodies: Vec<(Entity, BodyHandle)> = world
            .query::<PhysicsBody>()
            .map(|(e, b)| (e, b.rigid_body_handle))
            .collect();

        for (entity, handle) in bodies {
            if let Some(rb) = self.physics.rigid_body(handle) {
                let pos = rb.translation();
                if let Some(tr) = world.get_mut::<Transform>(entity) {
                    tr.position = Vec2::new(pos.x, pos.y);
                }
            }
        }
    }
}

fn run(&mut self, world: &mut World, _dt: f32) {
    if let Some(audio) = world.resource_mut::<AudioManager>() {
        // 파일 재생 (MP3 / OGG / WAV)
        audio.play("bgm", "assets/bgm.mp3", true);   // repeat=true → 무한 반복
        audio.play("sfx_jump", "assets/jump.wav", false);

        // 정지
        audio.stop("bgm");

        // 볼륨 (0.0 ~ 1.0)
        audio.set_volume("bgm", 0.5);

        audio.set_pan("sfx_enemy", 0.8);

}

  • set_pan은 다음 play() 호출부터 적용된다.

채널 재생 상태

AudioManager::playback_state로 채널이 없는지, 재생 중인지, 자연 종료됐는지 구분할 수 있다. stop(channel)은 채널을 제거하므로 Missing이 되고, 반복하지 않는 사운드가 끝까지 재생되면 Finished로 남는다.

use engine::AudioChannelState;

if let Some(audio) = world.resource_mut::<AudioManager>() {
    match audio.playback_state("bgm") {
        AudioChannelState::Missing => audio.play("bgm", "assets/track1.mp3", false),
        AudioChannelState::Playing => {}
        AudioChannelState::Finished => audio.play("bgm", "assets/track2.mp3", false),
    }

    if audio.is_finished("bgm") == Some(true) {
        audio.play("bgm", "assets/next_track.mp3", false);
    }
}

이펙트와 릴리스 엔벨로프

set_effect(channel, AudioEffect)로 채널별 이펙트를 설정한다. low_pass_hz(로우패스 필터) / pitch(재생 속도) / attack_secs(페이드 인)는 다음 play_* 호출부터 적용된다. release_secs > 0.0이면 stop(channel)이 즉시 끊는 대신 지정한 시간(초)에 걸쳐 볼륨을 0까지 페이드한 뒤 싱크를 제거한다(릴리스 엔벨로프). 페이드 진행에는 AudioSystem 등록(또는 수동 update(dt) 호출)이 필요하다.

use engine::AudioEffect;

audio.set_effect("bgm", AudioEffect { release_secs: 1.5, ..Default::default() });
audio.play("bgm", "assets/track1.mp3", true);
// ...
audio.stop("bgm");   // 1.5초 페이드 아웃 후 채널 제거
  • 진행 중인 stop_when_done 페이드(릴리스 또는 fade_out)가 있을 때 stop을 호출하면 즉시 끊는다 — fade_out은 릴리스 바이패스 경로다.
  • 같은 채널에 새 play_*를 호출하면 릴리스를 건너뛰고 즉시 새 소리를 재생한다.
  • 자연 종료(드레인)된 채널에 stop을 호출하면 릴리스 없이 즉시 제거된다.
  • 릴리스/페이드 완료 후 채널 볼륨은 set_volume으로 설정한 값으로 복원된다 (다음 재생이 무음으로 시작하지 않음).
  • release_secs == 0.0(기본값)이면 기존처럼 즉시 정지한다.

릴리스 엔벨로프 데모: cargo run --example audio_fades (R = 릴리스 사운드 시작, S = 릴리스 정지, I = 즉시 정지).

// 카메라를 플레이어 중심으로 이동

        }
    }
}

월드 바운드 클램핑

bounds: Option<(Vec2, Vec2)>(기본 None)를 설정하면 카메라 뷰포트가 월드 사각형 (min, max) 밖으로 스크롤되지 않도록 제한된다. App이 매 프레임 Camera::update() 직후 clamp_to_bounds()를 자동 호출하므로, 내장 follow를 쓰면 추가 코드가 필요 없다.

if let Some(cam) = world.resource_mut::<Camera>() {
    // 레벨 크기가 2000×1500이면 카메라가 그 밖으로 넘어가지 않는다
    cam.bounds = Some((Vec2::ZERO, Vec2::new(2000.0, 1500.0)));
}

// 카메라를 수동으로 옮긴 경우 직접 클램프할 수도 있다 (viewport 크기 전달).
// ViewportSize 불변 차용을 먼저 끝낸 뒤 Camera를 가변 차용한다.
let (vw, vh) = {
    let vp = world.resource::<ViewportSize>().unwrap();
    (vp.width, vp.height)
};
if let Some(cam) = world.resource_mut::<Camera>() {
    cam.clamp_to_bounds(vw, vh);
}
월드가 뷰포트보다 작은 축에서는(max - min < viewport / zoom) 클램프 범위가 뒤집히므로, 해당 축의 position은 bounds.min에 고정된다. 보이는 영역은 viewport / zoom이라 zoom도 반영된다.

에셋 서버

AssetServer 리소스가 이미지·스크립트·아틀라스를 로드·캐싱·핫리로딩한다. App::new()가 자동으로 삽입하므로 직접 생성할 필요 없다.

네이티브 빌드에서는 존재하는 파일 경로를 canonical path로 정규화해 캐시 키로 사용한다. 존재하지 않는 경로는 입력 문자열을 보존해 실패 상태와 마젠타 fallback을 유지하며, WASM에서는 URL/상대경로 의미를 보존하기 위해 정규화하지 않는다.

이미지 로드

// App 레벨 (run() 전)

// 시스템 내부
let assets = world.resource_mut::<AssetServer>().unwrap();
let handle = assets.load_image("assets/enemy.png");

// Sprite에 적용
world.add_component(e, Sprite { texture: Some(handle.path().into()), ..Default::default() });

같은 경로를 다시 호출하면 캐시된 핸들을 반환한다 (파일 I/O 없음).

저장/불러오기

engine::save 모듈은 RON 직렬화 데이터를 ChaCha20-Poly1305 AEAD로 암호화한 세이브 파일을 제공한다. serde::Serialize + serde::Deserialize를 구현한 임의의 타입을 저장·복원할 수 있다.

use engine::save::{save, load, load_or_default, exists, delete, save_path};
use serde::{Serialize, Deserialize};

#[derive(Serialize, Deserialize, Default)]
struct GameProgress {
    level: u32,
    score: u64,
}

// OS 표준 데이터 디렉토리 아래 경로 생성
let path = save_path("my_game", "progress.save");

// 저장
save(&path, &GameProgress { level: 3, score: 15000 }).unwrap();

// 불러오기
let progress: GameProgress = load(&path).unwrap();

// 파일 없으면 Default 반환
let progress: GameProgress = load_or_default(&path).unwrap();

// 존재 확인 / 삭제
if exists(&path) { delete(&path).unwrap(); }
함수설명
save(path, &T)RON 직렬화 후 암호화 저장. 부모 디렉토리 자동 생성
load(path)암호문을 복호화한 뒤 RON 역직렬화. 파일 없으면 SaveError::Io(NotFound), 변조/손상은 SaveError::Corrupted
load_or_default(path)파일 없으면 T::default() 반환. 복호화·파싱 오류는 전파
exists(path)파일 존재 여부 확인
delete(path)파일 삭제. 파일 없으면 Ok(())
save_path(app, file)OS 데이터 디렉토리 아래 경로 반환
WASM 패리티 (localStorage): save/load·save_versioned/load_migrated·write_ron/read_ron·exists/delete 전부 wasm에서 동작한다. AEAD 코어(ChaCha20-Poly1305)는 크로스플랫폼이고, wasm에서는 파일 대신 경로 문자열을 키로 localStorage에 값을 넣는다 — 암호화 블롭은 16진 인코딩되고, 논스 RNG는 OsRng(getrandom-js)다. localStorage는 사용자가 들여다볼 수 있으므로 내장 키는 비밀 유지가 아니라 난독화 + 변조 탐지용이다(네이티브 파일과 동일). 이 wasm AEAD localStorage 왕복은 예제 wasm_save + scripts/wasm_save_smoke.sh로 브라우저 검증된다(헤드리스 7/7).

경로 탐색 (A*)

engine::pathfinding 모듈은 격자 기반 A* 알고리즘을 제공한다.

PathGrid

통행 가능 여부를 저장하는 격자. row-major Vec<bool> 배열.

use engine::{PathGrid, find_path};
use glam::IVec2;

// 20×15 격자 생성 (전부 통행 가능)
let mut grid = PathGrid::new(20, 15);

// 장애물 설정
grid.set_walkable(5, 3, false);
grid.set_walkable(5, 4, false);
grid.set_walkable(5, 5, false);

// 통행 가능 여부 확인 (범위 밖이면 false, 패닉 없음)
let ok = grid.is_walkable(5, 3); // false
메서드설명
PathGrid::new(w, h)전부 통행 가능 상태로 초기화
PathGrid::new_blocked(w, h)전부 막힌 상태로 초기화
set_walkable(x, y, bool)특정 셀 통행 가능 여부 설정
is_walkable(x, y)통행 가능 여부 반환. 범위 밖 = false

find_path

let path: Option<Vec<IVec2>> = find_path(&grid, IVec2::new(0, 0), IVec2::new(19, 14));

match path {
    Some(steps) => {
        // steps: start 미포함, goal 포함
        for cell in &steps {
            println!("→ ({}, {})", cell.x, cell.y);
        }
    }
    None => println!("경로 없음"),
}
  • 4방향 이동 (상하좌우). 대각선 없음.
  • start == goal → Some(vec![goal])
  • 목표 셀이 막혀 있으면 즉시 None
  • 반환 경로는 start 미포함, goal 포함

타일맵과 연동 예

// Tilemap 크기에 맞춰 PathGrid 생성
let grid = PathGrid::new(tilemap.width as i32, tilemap.height as i32);
// ... 타일 종류에 따라 set_walkable 설정

// 적 AI: 매 프레임 또는 경로 갱신 주기마다 호출
if let Some(path) = find_path(&grid, enemy_tile, player_tile) {
    // path[0]이 다음 이동 목표 셀
}

// Tilemap 에서 직접 생성 (is_blocked 조건 지정)
let grid = PathGrid::from_tilemap(&tilemap, |id| id == 1);

대각선 경로탐색 (8방향 A*)

find_path_diagonal은 8방향(상하좌우 + 대각선) A* 알고리즘이다. 정수 비용(직교 = 10, 대각 = 14)과 옥타일 휴리스틱을 사용해 가장 짧은 경로를 찾는다.

use engine::{PathGrid, find_path_diagonal};
use glam::IVec2;

let grid = PathGrid::from_tilemap(&tilemap, |id| id != 0);
let path = find_path_diagonal(&grid, IVec2::new(0, 0), IVec2::new(10, 8));
  • 코너 컷 없음: 대각 이동 시 인접한 직교 두 셀이 모두 통행 가능해야 한다.
  • 반환 경로 규약은 find_path와 동일: start 미포함, goal 포함.
  • PathGrid::from_tilemap(&tilemap, is_blocked)로 Tilemap에서 직접 그리드를 생성할 수 있다 (tiles[y][x] → grid (x, y) 매핑).

컴포넌트설명
SteeringVelocity { velocity: Vec2, max_speed: f32 }스티어링 계산 결과 저장. SteeringSystem이 이 값을 읽어 Transform 이동
Seek { target: Vec2, max_speed: f32 }목표 위치를 향해 최대 속도로 직선 이동
Flee { target: Vec2, max_speed: f32, flee_radius: f32 }flee_radius 이내 접근 시 도망. 반경 밖이면 정지
Arrive { target: Vec2, max_speed: f32, slow_radius: f32, stop_radius: f32 }목표에 가까워질수록 감속, stop_radius 이내 정지
Wander { max_speed: f32, change_interval: f32 }change_interval마다 방향 변경하며 배회

SteeringSystem 등록

app.add_system(SteeringSystem);  // Seek→Flee→Arrive→Wander 순서 처리 후 Transform 이동

사용 예

// 적이 플레이어를 향해 이동
world.add_component(enemy, Seek { target: player_pos, max_speed: 120.0 });
world.add_component(enemy, SteeringVelocity::default());

// 근접 시 도망
world.add_component(minion, Flee { target: player_pos, max_speed: 150.0, flee_radius: 80.0 });
world.add_component(minion, SteeringVelocity::default());

// 목표 지점에 부드럽게 정착
world.add_component(unit, Arrive {
    target: destination,
    max_speed: 100.0,
    slow_radius: 60.0,
    stop_radius: 8.0,
});
world.add_component(unit, SteeringVelocity::default());
여러 스티어링 컴포넌트를 동시에 붙이면 마지막 처리된 것이 적용된다 (Seek→Flee→Arrive→Wander 우선순위).

Blackboard

Blackboard는 비헤이비어 트리 노드 간 공유 상태를 저장하는 독립 ECS 컴포넌트다.

API

메서드설명
Blackboard::new()빈 Blackboard 생성
set_bool(key, val) / get_bool(key) -> Option<bool>bool 값
set_float(key, val) / get_float(key) -> Option<f32>f32 값
set_int(key, val) / get_int(key) -> Option<i32>i32 값
set_vec2(key, val) / get_vec2(key) -> Option<Vec2>Vec2 값
set_string(key, val) / get_string(key) -> Option<&str>String 값

사용 예

// 엔티티에 Blackboard 부착
world.add_component(enemy, Blackboard::new());

// BehaviorNode::tick 내부에서 접근
fn tick(&mut self, world: &mut World, entity: Entity, _dt: f32) -> BehaviorStatus {
    if let Some(bb) = world.get_mut::<Blackboard>(entity) {
        let in_range = bb.get_bool("player_in_range").unwrap_or(false);
        bb.set_float("last_seen_time", 0.0);
    }
    BehaviorStatus::Success
}

CommandBuffer

Commands는 시스템 실행 중 엔티티/컴포넌트 변경을 안전하게 지연 예약하는 버퍼다.

쿼리 이터레이터가 살아있는 동안 world.spawn() 등을 직접 호출하면 borrow 충돌이 발생한다. Commands는 명령을 클로저로 큐에 쌓아두었다가 world.apply_commands(cmds) 로 일괄 적용한다.

API

메서드설명
Commands::new()빈 버퍼 생성
spawn(f)f(world, entity)를 apply 시 실행. 클로저 안에서 컴포넌트 추가
despawn(entity)엔티티 삭제 예약. 이미 없으면 noop
insert::<T>(entity, comp)컴포넌트 추가 예약
remove::<T>(entity)컴포넌트 제거 예약
world.apply_commands(cmds)큐 순서대로 일괄 적용

사용 예

struct SpawnSystem;
impl System for SpawnSystem {
    fn run(&mut self, world: &mut World, _dt: f32) {
        let mut cmds = Commands::new();

        // 쿼리 루프 중 삭제/생성 예약
        let dead: Vec<Entity> = world.query::<Health>()
            .filter(|(_, h)| h.0 <= 0)
            .map(|(e, _)| e)
            .collect();

        for e in dead {
            cmds.despawn(e);
            cmds.spawn(|world, ne| {
                world.add_component(ne, Particle::default());
            });
        }

        world.apply_commands(cmds);  // 루프 끝난 뒤 일괄 적용
    }
}

에디터 — 도킹 모드 (F2)

F2로 진입하는 두 번째 에디터 모드 (네이티브 전용, v7.1.0). 상용 엔진처럼 사이드 패널이 화면 가장자리를 차지하고 게임은 가운데 뷰포트에 표시되므로, 편집 UI가 게임 화면을 가리지 않는다.

  • 레이아웃: 상단 툴바(▶/⏸ 재생 제어, ⏭ 한 프레임 스텝, Snap, 씬 Save/Load) · 좌측 Entities/Scene 탭 · 우측 Inspector · 하단 Assets · 중앙 게임 뷰포트
  • F1 오버레이와 병행: F1은 기존 떠다니는 창 오버레이 그대로 (퀵 디버그용). 두 모드는 상호 배타 — F2 진입 시 오버레이 꺼짐, Docked에서 F1 누르면 오버레이로 전환
  • 뷰포트 입력: 중앙 뷰포트 안의 클릭/드래그/휠은 게임과 기즈모에 전달되고, 패널·팝업 위 입력은 egui가 가진다. Inspector에 타이핑하는 동안 게임 키 입력 차단
  • 일시정지: ⏸는 엔진 차원에서 씬 시스템 실행을 멈춘다 (GameState 리소스와 무관). 정지 중에도 기즈모 드래그 시 자식 엔티티가 따라온다 (HierarchySystem은 계속 실행). ⏭로 정확히 한 프레임 진행
  • 뷰포트 크기: 도킹 중 ViewportSize는 중앙 패널 크기를 보고하므로 카메라·스크린 UI·screen_to_world가 무수정 동작. 패널 크기를 조절하면 게임 뷰포트도 따라 변한다 (3프레임 디바운스)
WASM 빌드에서는 사용 불가 (#[cfg(not(target_arch = "wasm32"))]로 컴파일 제외).

에디터 — UI 레이아웃 편집 + 씬 영속화 (v8.0.0)

도킹/오버레이 에디터에서 UI 위젯과 스프라이트를 배치·크기 조절하고, 그 결과를 씬 RON 파일로 저장·로드할 수 있다.

  • 스크린 좌표 기즈모: UiNode를 가진 엔티티를 선택하면 화면 좌표에서 본체 드래그로 이동(offset), 모서리·변 8핸들로 크기 조절(size). 월드 스프라이트는 8핸들로 Transform.scale을 중심 고정 리사이즈. Snap 적용, Ctrl+Z 언두(EditorCmd::MoveUiNode/ResizeUiNode/ResizeEntity).
  • 위젯 직렬화: 모든 UI 위젯(UiNode, Button, Label, TextInput, Slider, CheckBox, ScrollView, Panel, LocalizedText)이 serde + Reflect 구현 — 인스펙터에서 편집되고 씬 파일에 저장된다. 런타임 상태(커서/value/scroll_offset 등)는 직렬화에서 제외, 설계값은 TextInput.initial_text/Slider.initial_value로 저장된다.
  • 컴포넌트 직렬화 레지스트리: App::register_serde_component::<T>(name, post_spawn)로 임의의 Serialize + DeserializeOwned + Clone 컴포넌트를 씬 저장/로드 대상에 등록한다. UI 위젯은 자동 등록되며, 게임은 자체 컴포넌트(예: 스탯)를 같은 방식으로 등록한다. 로드 시 미등록 컴포넌트 이름은 경고 후 건너뛴다(로드는 실패하지 않음). EntityDef에 components 맵이 추가되고 SceneDef 버전이 3으로 올라갔다(v2 파일 하위호환).
  • 예제: cargo run --example ui_layout_editor_game — 기본 메뉴를 F2 에디터로 배치·리사이즈 → Save Scene → 재시작 시 저장본 로드(레이아웃 편집 인수 테스트).
네이티브 전용. 씬 파일 저장/로드 자체는 save.rs 계층에서 이미 wasm 게이트됨.

에디터 — 게임 데이터 편집 (v8.1.0)

컴포넌트 스탯과 RON 데이터 테이블을 에디터에서 편집하고 파일로 영속화한다.

  • #[derive(Reflect)] (워크스페이스 크레이트 engine_reflect_derive): f32/i32/Vec2/bool/String/Color/[f32;4] 필드 구조체의 Reflect 구현을 자동 생성. #[reflect(skip)]로 필드 제외, 미지원 타입은 컴파일 에러. 손수 쓴 impl과 공존. 게임은 engine처럼 이 크레이트를 직접 의존성에 추가하고 use engine_reflect_derive::Reflect; 후 #[derive(Reflect)]. (엔진에서 re-export하지 않는 이유: 미게시 proc-macro에 의존하면 skeleton-engine 자체가 crates.io에 publish되지 않으므로 별도 크레이트로 분리.)
  • App::register_editable_component::<T>(name, post_spawn): 한 번 호출로 인스펙터 편집(Reflect) + 복제(Clone) + 씬 저장/로드(serde) + Add/Remove 컴포넌트 버튼을 전부 등록. T: Reflect + Serialize + Deserialize + Clone + Default.
  • 데이터 테이블(DataTable, DataTableRegistry, App::load_data_table): 스키마 무지 RON 표([(col: value, …), …])를 로드해 런타임에 World 리소스로 읽고, 에디터 하단의 Data Tables 탭에서 셀 단위(숫자/문자열/불) 편집·행 추가·삭제·저장하며, 디스크 변경을 실행 중 게임에 핫리로드한다(미저장 편집은 dirty 가드로 보호). 패널은 네이티브 전용, 타입은 크로스플랫폼.
  • 예제: cargo run --example stat_editor_game — derive한 Stats 컴포넌트를 enemies 테이블에서 시드, 인스펙터에서 스탯 편집(HUD 즉시 반영) + Data Tables 패널에서 enemies/items 표 튜닝(게임 데이터 편집 인수 테스트).

engine::scripting 모듈은 Rhai 스크립트 파일을 에셋으로 로드하고 엔티티에 부착해 실행한다. 신뢰된 로컬 게임 스크립트용이며, 적대적 사용자 입력을 격리하는 sandbox 경계가 아니다.

// 엔티티에 ScriptRunner 부착

Commands

Steering

스티어링 함수는 상호 배타적이다 — 하나를 호출하면 다른 스티어링 컴포넌트가 엔티티에서 제거된다 (마지막 호출이 이긴다). Rust 코드에서 컴포넌트를 직접 조합하는 경로는 영향받지 않는다.

스크립트 예시

// enemy_ai.rhai
let in_range = bb_get_bool("player_in_range");
if in_range {
    seek_target(player_x, player_y, 120.0);
} else {
    stop_steering();
    bb_set_float("idle_time", bb_get_float("idle_time") + dt);
}

    → 스왑체인 (화면 출력)

라이팅만 활성화된 경우:

스프라이트 렌더링
    → 중간 씬 텍스처 (scene_texture_for_lighting)
    → LightingRenderer.run_pass (라이팅 적용)
    → 스왑체인 (화면 출력)
플랫폼: 라이팅은 네이티브(native) 전용. WASM 빌드에서는 비활성화된다.

이번 프레임에 추가/교체된 컴포넌트만 조회할 수 있다. dirty-flag 패턴을 대체한다.

API

    println!("새 엔티티 {:?}: HP={}", entity, health.0);

// 이번 프레임에 교체되거나 명시적으로 changed 표시된 컴포넌트를 가진 엔티티
for (entity, transform) in world.query_changed::<Transform>() {
    update_spatial_cache(entity, transform.position);

// 필드를 직접 수정한 경우에는 changed를 명시 표시
if let Some(transform) = world.get_mut::<Transform>(entity) {

// 명시적 초기화 (App이 매 프레임 자동 호출 — 직접 호출 불필요)
world.clear_change_tracking();

Flat-normal 2D lighting

v2.0 removes the misleading public sprite normal-map API. The lighting renderer still keeps an internal flat-normal buffer initialized to [0.5, 0.5, 1.0] every frame, so PointLight and AmbientLight continue to work without per-sprite normal textures.

PointLight.light_height

Virtual Z height for a point light. Lower values make lighting more directional; higher values behave closer to frontal light. This works against the flat normal buffer.

world.add_component(light_entity, PointLight {
    color: [1.0, 0.9, 0.7],
    radius: 300.0,
    intensity: 2.0,
    light_height: 0.15,
});

스프라이트 렌더링 (scene_tex)
    ↓
노멀 버퍼 초기화 (flat normal [0.5, 0.5, 1.0])
    ↓
LightingRenderer (scene_tex + normal_buf → Lambert diffuse)
    ↓
화면 출력

Camera 리소스에 세 가지 이펙트가 내장된다. App이 매 프레임 자동으로 camera.update(dt, follow_pos)를 호출하므로 별도 시스템 등록 불필요.

Camera Shake

// 시스템 내부에서

오브젝트 풀

총알·파티클처럼 자주 생성/소멸되는 엔티티를 재활용한다. 반납된 엔티티는 Pooled 마커 컴포넌트로 표시되며, query_without::<Pooled>() 로 시스템에서 제외할 수 있다.

Pool API

// 리소스로 등록 (최대 64개 재사용)
world.insert_resource(Pool::new(64));

// 획득 — 풀에 있으면 재사용, 없으면 새 스폰
let pool = world.resource_mut::<Pool>().unwrap();
let bullet = pool.acquire(&mut world, |w, e| {
    w.add_component(e, Transform { position: spawn_pos, ..Default::default() });
    w.add_component(e, Bullet { speed: 400.0 });
    w.add_component(e, Sprite::colored(1.0, 1.0, 0.0));
});

// 반납 — 풀에 저장 (capacity 초과 시 despawn)
pool.release(bullet, &mut world);
메서드설명
Pool::new(capacity)최대 capacity개 엔티티를 저장하는 풀 생성
acquire(world, setup)풀에서 엔티티 가져오기. 없으면 spawn. setup 클로저로 초기화
release(entity, world)엔티티 반납. Pooled 마커 추가. 풀 초과 시 despawn
available_count()현재 대기 중인 엔티티 수
capacity()최대 풀 용량
clear(world)풀 전체 비우기 + despawn

        // 발사

            if let Some(mut pool) = world.take_resource::<Pool>() {
                let pos = player_transform.position;
                pool.acquire(world, move |w, e| {
                    w.add_component(e, Transform { position: pos, ..Default::default() });
                    w.add_component(e, Bullet { velocity: Vec2::new(0.0, -400.0) });
                });
                world.insert_resource(pool);
            }
        }

        // 화면 밖 총알 회수

            .query_without::<Bullet, Pooled>()
            .filter(|(_, b)| b.position.y < -100.0)

                }
                world.insert_resource(pool);
            }
        }
    }
}

시스템 실행 순서 (Phase 45)

add_system만 사용하면 등록 순서대로 실행된다. add_system_labeled와 SystemConfig를 사용하면 라벨 기반 위상정렬로 실행 순서를 명시적으로 지정할 수 있다.

SystemLabel / SystemConfig

pub type SystemLabel = &'static str;

SystemConfig는 빌더 패턴. SystemConfig::new()로 시작해 원하는 제약을 체인한다.

메서드설명
.label(l)이 시스템에 라벨을 붙인다. 다른 시스템이 before/after로 참조 가능
.before(l)라벨 l의 시스템보다 앞에 실행되도록 요청
.after(l)라벨 l의 시스템보다 뒤에 실행되도록 요청
.in_set(s)SystemSet s에 배치. 해당 set이 비활성화되면 실행 건너뜀

App 메서드

// 라벨/순서/그룹 설정과 함께 시스템 등록
app.add_system_labeled(system, config);

// SystemSet 활성/비활성. 비활성 set의 시스템은 매 프레임 실행에서 제외됨
app.set_enabled(set: SystemLabel, enabled: bool);

// 순환 의존성/시스템 panic 처리 정책을 엄격하게 바꿀 수 있음
app.set_schedule_error_policy(ScheduleErrorPolicy::PanicOnCycle);
app.set_system_panic_policy(SystemPanicPolicy::AbortAfterLog);

순서 규칙

  • after("X") → "X" 라벨을 가진 모든 시스템 뒤에 배치
  • before("Y") → "Y" 라벨을 가진 모든 시스템 앞에 배치
  • 동순위(제약 없는 것들)는 삽입 순서로 결정됨 (결정적)
  • 순환 의존성 발생 시 기본 정책은 삽입 순서 폴백 + 에러 로그
  • ScheduleErrorPolicy로 순환 시 사용자 시스템 실행 건너뛰기 또는 panic을 선택할 수 있음
  • SystemPanicPolicy 기본값은 panic 시스템 비활성화 후 계속 실행. panic = "abort" 빌드에서는 unwind가 없어 이 복구가 적용되지 않음

물리 조인트

PhysicsWorld에서 두 바디를 연결하는 조인트를 생성한다. 반환된 JointHandle로 나중에 제거할 수 있다. 예제: examples/crane_wrecking_ball.

use engine::{PhysicsWorld, JointHandle};

if let Some(physics) = world.resource_mut::<PhysicsWorld>() {
    // 회전 조인트 (힌지) — 두 바디가 공유 축 주위를 자유 회전
    let hinge: JointHandle = physics.add_revolute_joint(
        body1, body2,
        Vec2::new(0.0, -16.0),  // body1 로컬 앵커
        Vec2::new(0.0,  16.0),  // body2 로컬 앵커
    );

    // 거리 조인트 (스프링) — rest_length 거리 유지
    let rope: JointHandle = physics.add_distance_joint(
        body1, body2,
        Vec2::ZERO, Vec2::ZERO,
        80.0, // rest_length (픽셀)
    );

    // 프리즘 조인트 (슬라이더) — 지정 축을 따라서만 이동 허용
    let slider: JointHandle = physics.add_prismatic_joint(
        body1, body2,
        Vec2::ZERO, Vec2::ZERO,
        Vec2::X, // 이동 허용 축 (정규화됨)
    );

    // 제거
    physics.remove_joint(hinge);
}
물리 기능은 네이티브 전용이다. WASM 타겟에서는 #[cfg(not(target_arch = "wasm32"))]로 감싸야 한다.

DrawText — 멀티라인 + 정렬

DrawText의 align과 bounds 필드를 활용해 멀티라인 정렬 텍스트를 렌더링한다.

#[derive(Debug, Clone)]
pub struct DrawText {
    pub text: String,
    pub position: Vec2,       // 좌상단 픽셀 좌표 (스크린 스페이스)
    pub bounds: Option<Vec2>, // 텍스트 레이아웃 영역. None이면 화면 끝까지
    pub size: f32,            // 폰트 픽셀 크기

}

저장 마이그레이션

세이브 파일 스키마가 바뀔 때 SaveMigrator로 이전 버전을 새 버전으로 단계적으로 업그레이드한다. 네이티브 전용 (cfg(not(target_arch = "wasm32"))).

use engine::{save_versioned, load_migrated, SaveMigrator};

// 저장: 현재 스키마 버전 태그를 함께 기록
save_versioned(&path, 1, &my_save_data)?;

// 로드: 저장 버전에서 현재 버전까지 단계적 마이그레이션
let migrator = SaveMigrator::new()
    // 버전 0 → 1: "coins" 필드 추가
    .step(0, |mut v| {
        if let ron::Value::Map(ref mut m) = v {
            m.insert(
                ron::Value::String("coins".into()),
                ron::Value::Number(ron::value::Number::new(0i64)),
            );
        }
        v
    });
// migrator.current_version() == 1

let data: MySave = load_migrated(&path, &migrator)?;
  • 저장된 버전이 migrator.current_version()보다 크면 (미래 버전 세이브) → SaveError::Unsupported
  • 같은 버전이면 마이그레이션 없이 역직렬화
  • 커스텀 키: save_versioned_with_key / load_migrated_with_key

저장 암호화 (Phase 53)

save / load는 기존 시그니처를 유지하며 기본 키로 자동 암호화한다. 앱별 키가 필요하면 SaveKey, save_with_key, load_with_key를 사용한다.

암호화 방식

  • 알고리즘: ChaCha20-Poly1305 AEAD (chacha20poly1305 크레이트)
  • 파일 포맷: [매직 9B] [Nonce 12B] [암호문 + 인증태그]
  • Nonce는 저장할 때마다 랜덤 생성 → 같은 데이터도 매번 다른 파일이 생성됨

변조 감지

복호화 인증 실패(파일 변조 또는 손상)는 SaveError::Corrupted를 반환한다.

#[derive(Debug)]
pub enum SaveError {
    Io(io::Error),      // 파일 읽기/쓰기 실패
    Ron(String),        // 직렬화/역직렬화 실패
    Corrupted,          // 복호화 인증 실패 — 변조 또는 손상
}

API

use engine::save::{
    save, load, load_or_default, save_with_key, load_with_key,
    SaveError, SaveKey, exists, delete, save_path,
};

// OS 표준 데이터 디렉토리 경로 계산

// 저장 (자동 암호화)

// 로드 (자동 복호화)

// 없으면 default 반환, 변조 등 에러는 그대로 전파

// 파일 존재 여부

// 파일 삭제 (없어도 Ok)
delete(&path)?;

사용 예

암호화를 위해 기존 코드를 변경할 필요가 없다. save/load를 그대로 쓰면 자동으로 암호화된 파일이 생성되고 복호화되어 읽힌다.

#[derive(Serialize, Deserialize, Default)]

    level: u32,
    score: u64,
    unlocked_items: Vec<String>,
}

// 저장

RemoteEntities<K>

서버가 소유한 원격 엔티티의 id → Entity 생명주기(처음 보이면 스폰, 사라지면 despawn)를 관리한다. 모든 네트워크 게임이 인라인으로 반복하던 HashMap<id, Entity> 패턴을 대체한다. 매핑/스폰/디스폰만 담당하고, 무엇을 스폰할지(클로저)·기존 엔티티 갱신(get 후 World로 변경)·게임 상태 맵은 게임 쪽에 남긴다.

use engine::RemoteEntities;

let mut remotes: RemoteEntities<usize> = RemoteEntities::new();

// 위치 갱신 수신 시: 처음이면 스폰, 이후엔 기존 Entity 갱신
let e = remotes.get_or_spawn(world, id, |w| w.spawn());
if let Some(tr) = world.get_mut::<Transform>(e) {
    tr.position = new_pos;
}

remotes.remove(world, &id); // "bye" 수신 시: 제거 + despawn
remotes.clear(world);       // 연결 끊김 / 씬 리셋 시: 전체 despawn
메서드설명
get_or_spawn(world, key, spawn)key의 Entity 반환. 처음이면 spawn 클로저로 생성 후 삽입
get(&key)현재 매핑된 Entity (Option)
contains_key(&key)트래킹 여부
remove(world, &key)제거 + 해당 Entity despawn (없으면 no-op)
clear(world)전체 despawn + 맵 비우기
len() / is_empty() / iter()개수 / 빈 여부 / (&key, Entity) 반복
의도적으로 최소 구현. 생명주기 매핑만 담당한다. 상태 보간은 별도의 SnapshotBuffer<T>로 분리되어 있고(아래 참고), 클라이언트 예측·업데이트 콜백·타입드 엔티티 등 더 풍부한 기능은 각각을 실제로 요구하는 예제가 나올 때까지 보류한다. docs/REMOTE_ENTITIES_DESIGN.md 참고.

예제: examples/games/coin_race/(권위 서버 + 클라이언트, 네이티브·wasm), examples/mp_client.rs / mp_server.rs(위치 릴레이).

SnapshotBuffer<T: Lerp>

원격 값을 낮은 스냅샷 레이트(예: 10–30 Hz)로 받아도 60 Hz로 부드럽게 렌더하기 위한 보간 버퍼. 스냅샷이 도착할 때마다 클라이언트 단조 시계로 타임스탬프를 찍어 push하고, 매 프레임 약간 과거 시점(now - interp_delay)을 sample하면 항상 실제 샘플 두 개 사이를 보간한다(범위를 벗어나면 양 끝으로 클램프). Lerp를 구현한 모든 T에 대해 제네릭이라 f32(회전각)·Vec2(위치)·Color 등 무엇이든 보간할 수 있다.

RemoteEntities와는 직교한다: 생명주기(id → Entity)는 RemoteEntities가, 렌더러가 읽는 값 히스토리는 SnapshotBuffer가 소유하며, 게임은 둘을 병렬 맵으로 둔다.

use engine::SnapshotBuffer;
use glam::Vec2;

// 엔티티 id별로 위치·회전 버퍼를 따로 둔다.
let mut pos: HashMap<usize, SnapshotBuffer<Vec2>> = HashMap::new();
let mut rot: HashMap<usize, SnapshotBuffer<f32>> = HashMap::new();

// 스냅샷 수신 시: 클라이언트 시계로 타임스탬프를 찍어 기록
pos.entry(id).or_default().push(client_time, Vec2::new(x, y));
rot.entry(id).or_default().push(client_time, angle);

// 렌더 시: interp_delay 만큼 과거를 샘플 → 두 스냅샷 사이를 lerp
let rt = client_time - interp_delay;
if let Some(p) = pos.get(&id).and_then(|b| b.sample(rt)) {
    transform.position = p;
}
메서드설명
new() / with_capacity(n)빈 버퍼 생성 (기본 8개 / 최근 n개 유지, 최소 2)
push(t, value)클라이언트 시각 t(초)로 스냅샷 기록. 역순·중복 타임스탬프는 무시
sample(rt)rt 시점의 보간 값(Option<T>). 범위 밖이면 양 끝 클램프, 비어 있으면 None
is_empty() / len()빈 여부 / 버퍼된 스냅샷 개수
interp_delay는 느낌 파라미터다. 너무 작으면 샘플이 모자라 끊기고, 너무 크면 시각 지연이 커진다. 스냅샷 간격의 1–2배가 대체로 무난하다(predict_shooter는 60 ms, orbital_dodger는 120 ms를 기본값으로 쓰며 런타임에 [/]로 조정). orbital_dodger에서 I로 보간을 끄면 원본 스냅샷의 끊김이 그대로 보인다.

예제: examples/games/orbital_dodger/(보간 전용 — 위치 Vec2 + 회전 f32 두 채널), examples/games/predict_shooter/(예측 + 원격 보간).

다이얼로그 (Dialogue)

DialogueBox는 RPG·비주얼 노벨·내러티브 게임이 매번 다시 구현하는 "한 글자씩 드러나는 대사 상자"를 재사용 가능한 컴포넌트로 만든 것입니다. 엔티티에 DialogueBox를 붙이고 DialogueSystem을 추가하면, 시스템이 타이프라이터 효과를 진행시키고 화면 하단에 화자 이름 + 드러난 텍스트를 그립니다(TextQueue로 스크린 공간 렌더링). 시스템은 입력에 무관(input-agnostic)합니다 — 언제 다음으로 넘어갈지는 게임이 직접 DialogueBox::advance()(예: Space 키)를 호출해 결정합니다. 기능은 점진적으로 쌓입니다: 분기 선택지, RON 데이터 기반 트리, 조건/효과가 달린 변수, 현지화, 초상화까지. 예제: dialogue_demo, dialogue_branching, dialogue_portrait, dialogue_quest.

기본 — 타이프라이터

DialogueBox::new(speaker, lines)로 만들고 DialogueSystem을 추가합니다(기본 28 chars/sec). advance()는 2단계로 동작합니다: 첫 호출은 현재 줄의 reveal을 즉시 완료하고, 다 드러난 뒤 호출하면 다음 줄로 넘어갑니다(마지막 줄 뒤엔 종료). 예제 dialogue_demo.

use engine::{App, DialogueBox, DialogueSystem};

let mut app = App::new();
let e = app.world.spawn();
app.world.add_component(
    e,
    DialogueBox::new("Guide", ["Welcome, traveler.", "Press space to continue."])
        .with_chars_per_sec(32.0),
);
app.add_system(DialogueSystem);

// 게임 자신의 시스템에서(예: Space 입력 시):
// if let Some(d) = world.get_mut::<DialogueBox>(e) { d.advance(); }
메서드 / 필드설명
DialogueBox::new(speaker, lines)화자 + 줄 목록으로 생성(기본 28 chars/sec).
.with_chars_per_sec(cps)타이프라이터 속도(빌더). <= 0.0 = 즉시 전체 표시.
advance()2단계: 첫 호출 = 현재 줄 reveal 완료, 다음 호출 = 다음 줄. 선택지 대기 중이면 no-op.
visible_text()지금까지 드러난 현재 줄의 부분 텍스트(UTF-8 안전).
line_fully_revealed()현재 줄이 전부 드러났는지.

분기 선택지

with_choices(line, choices)로 특정 줄이 다 드러난 뒤 보여줄 DialogueChoice 목록을 붙입니다. choose(i)는 i번째 선택지의 goto 줄 인덱스로 점프합니다(범위 밖 goto는 끝으로 clamp되어 대화 종료 — panic 없음). 선택지가 대기 중일 때 advance()는 no-op이라 plain advance로 결정을 건너뛸 수 없습니다. 두 분기를 같은 작별 줄로 reconverge시킬 수도 있습니다. 예제 dialogue_branching.

use engine::{DialogueBox, DialogueChoice};

let dbox = DialogueBox::new("Merchant", ["Welcome.", "Buy or leave?", "A fine choice.", "Farewell."])
    .with_choices(1, [
        DialogueChoice::new("Buy",   2),  // → 줄 2
        DialogueChoice::new("Leave", 3),  // → 줄 3
    ])
    .with_choices(2, [DialogueChoice::new("Go", 3)]); // reconverge

// 입력 시: 1/2 키로 d.choose(0) / d.choose(1)
// `pending_choices()`가 Some이면 결정 대기 중 — `choose`로 풀어야 advance가 진행됨.
메서드설명
.with_choices(line, choices)line 줄이 다 드러나면 보여줄 선택지(빌더, chainable).
pending_choices()지금 제시할 무조건(cond 없는) 선택지들의 Some(Vec<&DialogueChoice>), 없으면 None.
choose(i)pending_choices()의 i번째를 골라 goto로 점프. 범위 밖 인덱스/goto는 안전한 no-op/clamp.
DialogueChoice::new(text, goto)리터럴 라벨 + 점프 줄 인덱스 선택지.

데이터 기반 — RON 트리

DialogueTree는 분기 대화를 위한 authoring 포맷입니다: 이름 붙은 노드들의 순서 목록, 각 노드는 줄(리터럴 또는 현지화 키)과 goto를 노드 id로 참조하는 선택지를 가집니다. spawn 시 평범한 DialogueBox로 flatten되므로(노드 순서 = 줄 인덱스, 모든 goto id가 줄 인덱스로 해석됨) 기존 DialogueSystem이 그대로 구동합니다. App::load_dialogue(name, path)로 로드+등록하면 DialogueRegistry에 저장되고, 네이티브에서는 파일 변경이 핫리로드됩니다. 예제 dialogue_quest + dialogue_quest.dlg.ron.

use engine::{App, DialogueRegistry, DialogueSystem};

let mut app = App::new();
// "quest"라는 이름으로 트리를 로드(네이티브: 핫리로드 등록).
app.load_dialogue("quest", "examples/dialogue_quest.dlg.ron");

// 레지스트리에서 새 DialogueBox를 만들어 엔티티에 부착.
let dbox = app.world.resource::<DialogueRegistry>()
    .and_then(|r| r.box_of("quest"))
    .expect("tree loaded");
let e = app.world.spawn();
app.world.add_component(e, dbox);
app.add_system(DialogueSystem);
// dialogue_quest.dlg.ron — 노드 순서 = 줄 인덱스, goto는 노드 id로 참조
(
    speaker_key: "npc.merchant",
    chars_per_sec: 34.0,
    nodes: [
        (id: "intro", line_key: "dlg.intro"),
        (id: "ask", line_key: "dlg.ask", choices: [
            (key: "choice.buy",   goto: "buy", effect: EmitEvent(name: "grant_lantern")),
            (key: "choice.leave", goto: "bye"),
        ]),
        (id: "buy", line_key: "dlg.buy", choices: [(key: "choice.more", goto: "ask2")]),
        (id: "ask2", line_key: "dlg.ask2", choices: [
            (key: "choice.secret", goto: "secret",
             cond: (var: "has_lantern", op: Eq, value: Bool(true))), // 조건 게이트
            (key: "choice.leave", goto: "bye"),
        ]),
        (id: "secret", line_key: "dlg.secret", choices: [(key: "choice.go", goto: "bye")]),
        (id: "bye", line_key: "dlg.bye"),
    ],
)

리터럴 트리는 speaker_key/line_key/key 대신 speaker/line/text를 씁니다. 한 트리에 리터럴과 현지화 줄을 섞으면 파싱 에러입니다. cond/effect는 RON의 IMPLICIT_SOME 확장으로 Some(...) 없이 inline 작성됩니다.

API설명
App::load_dialogue(name, path)RON 트리를 로드해 name으로 등록(레지스트리 lazy 삽입 + 네이티브 핫리로드, scene reset에도 보존). wasm에서는 no-op.
DialogueTree::from_ron_str(s) / load(path)RON 텍스트/파일을 파싱·검증(중복 id, unknown goto, 리터럴/현지화 혼용을 거부).
DialogueTree::to_box() / node_ids()첫 노드에서 시작하는 새 DialogueBox 생성(clone) / 선언 순서 노드 id 목록.

조건 / 효과 — 변수

선택지는 변수에 따라 보이거나 숨겨지고(조건), 선택 시 부수 효과를 낼 수 있습니다. 변수는 게임 소유의 DialogueVars World 리소스에 삽입합니다. DialogueChoice.cond: DialogueCond(Eq/Ne/Gt/Lt/Ge/Le)는 가시성을 게이트하고, .effect: DialogueEffect는 SetVar(변수 설정)나 EmitEvent(Events<DialogueEvent>로 전송)를 적용합니다. 설정되지 않은 변수는 value 타입의 zero값으로 취급됩니다(예: 미설정 bool은 Eq Bool(false)에 통과).

중요 — 어떤 advance/choose가 변수를 존중하는가: plain DialogueBox::advance()/choose()는 변수를 모르며 무조건 선택지만 봅니다(전부 조건부인 줄에서 데드락하지 않도록). 조건 게이트와 효과를 적용하려면 게임 코드에서 world 레벨 헬퍼 dialogue::advance(world, entity) / dialogue::choose(world, entity, visible_index)를 호출하세요 — 이들이 DialogueVars를 읽어 선택지를 필터링하고, 고른 선택지의 효과(SetVar 쓰기 / EmitEvent 전송)를 적용합니다. visible_index는 visible_choices(&vars) 슬라이스 기준입니다. 예제 dialogue_quest.

use engine::{dialogue, App, DialogueEvent, DialogueVars};

let mut app = App::new();
app.world.insert_resource(DialogueVars::default()); // 게임 소유 — 시작 플래그 seed
app.register_event::<DialogueEvent>();              // EmitEvent 버스 등록

// 입력 시스템에서: 변수 인지(조건/효과 존중) advance/choose
// dialogue::advance(world, e);          // is_choosing 동안만 블록
// dialogue::choose(world, e, 0);        // visible 0번 선택 → goto + 효과 적용

// EmitEvent를 다른 시스템에서 읽어 게임 의미를 부여:
// for ev in world.resource::<Events<DialogueEvent>>().unwrap().read() {
//     if ev.name == "grant_lantern" { /* has_lantern = true */ }
// }
API설명
DialogueVars이름 붙은 변수 저장 리소스. set_bool/int/float/str, get_bool/int/float/str, clear.
DialogueCond { var, op, value }DialogueOp(Eq/Ne/Gt/Lt/Ge/Le)로 변수 비교. 순서 연산은 Int/Float만(비숫자는 false).
DialogueValueBool/Int/Float/Str 변수 값 enum.
DialogueEffectSetVar { key, value } 또는 EmitEvent { name }.

dialogue::advance(world, e)변수 인지 advance — is_choosing 동안만 블록(전부 게이트된 줄은 정상 진행).

현지화

DialogueBox::localized(speaker_key, line_keys)로 번역 키에서 상자를 만들면, DialogueSystem이 매 프레임 LocaleResource로 키를 해석합니다. lines/speaker는 비어서 시작했다가 첫 resolve에서 채워지므로, 대화 도중 로케일을 live 전환하면 reveal 진행을 잃지 않고 화자·줄·선택지가 모두 재번역됩니다. resolve(&locale)는 수동 호출도 가능하며 current/reveal 상태를 건드리지 않아 매 프레임 안전합니다. 선택지는 DialogueChoice::localized(key, goto)로 키를 갖습니다. 예제 dialogue_branching(EN↔KO 라이브 토글).

use engine::{DialogueBox, DialogueChoice};

let dbox = DialogueBox::localized(
        "npc.merchant",
        ["dlg.intro", "dlg.ask", "dlg.buy", "dlg.dark", "dlg.bye"],
    )
    .with_chars_per_sec(34.0)
    .with_choices(1, [
        DialogueChoice::localized("choice.buy",  2),
        DialogueChoice::localized("choice.dark", 3),
    ]);
// LocaleResource를 world에 삽입하면 DialogueSystem이 매 프레임 resolve.
// 게임에서 locale.set_locale("ko") 하면 다음 프레임에 라이브 재번역.

초상화

with_portrait(handle)로 화자 초상화(Handle<ImageAsset>)를 설정하면, DialogueSystem이 텍스트 왼쪽에 이미지를 그리고(UI 이미지 큐) 텍스트를 오른쪽으로 밀어 비켜줍니다. 초상화가 없는 상자는 이전과 byte-identical하게(원래 좌측 마진) 렌더링됩니다 — 추가적 기능입니다. 초상화는 런타임 전용이라 scene RON에 직렬화되지 않습니다. 예제 dialogue_portrait.

use engine::DialogueBox;

let sage = app.load_image("examples/assets/portrait_sage.png");
let dbox = DialogueBox::new("Sage", ["A portrait rides along with each line now."])
    .with_chars_per_sec(32.0)

타일맵 프로젝션 — 아이소메트릭 / 헥사고날

기본 Tilemap은 정사각 격자(직교, orthographic)지만, TilemapProjection을 지정하면 같은 tiles[row][col] 배열을 아이소메트릭 다이아몬드나 헥사고날 격자로 배치할 수 있습니다. 투영은 Tilemap::new(...) 뒤에 .with_projection(...) 빌더로 붙이며, TilemapSystem이 투영에 맞게 셀의 월드 좌표·렌더 크기·깊이(z)를 자동으로 계산합니다. 데이터 모델(tiles 배열)은 투영과 무관하게 동일하므로, 게임 로직은 항상 (row, col) 격자 좌표로 다루고 화면 배치만 투영이 담당합니다.

투영 변형격자좌표계깊이 z
Orthographic (기본)정사각—고정 -1.0 (겹침 없음)
Isometric2:1 다이아몬드 (폭 tile_size, 높이 tile_size/2)—row + col (뒤→앞 깊이 정렬)
Hexagonal뾰족머리(pointy-top) 육각형odd-r 오프셋 (홀수 행이 오른쪽으로 반칸)고정 -1.0 (겹침 없음)
HexagonalFlat납작머리(flat-top) 육각형odd-q 오프셋 (홀수 열이 아래로 반칸)고정 -1.0 (겹침 없음)

TilemapProjection은 #[non_exhaustive]이며, cell_center_world(row, col)(격자→월드 중심)와 그 역인 cell_at_world(world_pos)(월드→(row, col), 격자 밖이면 None)가 모든 투영을 일관되게 처리합니다.

아이소메트릭 (2:1 다이아몬드)

한 타일이 폭 tile_size, 높이 tile_size/2인 다이아몬드로 그려집니다. col이 늘면 오른쪽·아래로, row가 늘면 왼쪽·아래로 이동하며, 셀 (0, 0)의 중심이 origin에 놓입니다. 다이아몬드는 서로 겹치므로 TilemapSystem이 셀별 렌더 z = row + col로 뒤→앞 깊이 정렬합니다. 바닥 위에 서야 하는 엔티티(캐릭터 등)는 점유 셀보다 큰 z를 주세요. 픽킹은 다이아몬드 변환의 역함수로 처리됩니다. (예제 iso_tilemap)

use engine::{Tilemap, TilemapAtlas, TilemapProjection, TilemapSystem, Vec2};

let atlas = TilemapAtlas::new("examples/assets/iso_tiles.png", 2, 1);
// origin = 셀 (0,0)의 중심 위치
let tilemap = Tilemap::new(atlas, tiles, 64.0, Vec2::new(450.0, 150.0))
    .with_projection(TilemapProjection::Isometric);

// 마우스 픽킹: 커서 월드 좌표 → 셀 (역 다이아몬드 변환)
if let Some((row, col)) = tilemap.cell_at_world(cursor_world) {
    // hovered cell
}

헥사고날 (뾰족머리, odd-r)

뾰족머리 육각 격자를 odd-r 오프셋 좌표로 배치합니다(홀수 행이 오른쪽으로 반칸 밀림). tile_size는 육각형의 평행면-간(flat-to-flat) 폭이며, 셀 (0, 0)의 중심이 origin에 놓입니다. 뾰족머리 육각형은 폭보다 높이가 커서(cell_render_size가 tile_size × tile_size·2/√3 반환) 더 키 큰 스프라이트로 그려지며, 투명한 모서리는 이웃과 무해하게 겹칩니다. 육각형은 빈틈없이 맞물리므로 직교처럼 고정 z = -1.0을 씁니다. 픽킹은 픽셀→축(axial)→큐브-라운딩으로 처리됩니다. (예제 hex_tilemap)

let atlas = TilemapAtlas::new("examples/assets/hex_tiles.png", 2, 1);
// tile_size = 육각형 flat-to-flat 폭
let tilemap = Tilemap::new(atlas, tiles, 72.0, Vec2::new(150.0, 110.0))
    .with_projection(TilemapProjection::Hexagonal);

헥사고날 (납작머리, odd-q)

뾰족머리의 90° 회전 짝으로, 납작머리 육각 격자를 odd-q 오프셋 좌표로 배치합니다(홀수 열이 아래로 반칸 밀림). tile_size는 육각형의 평행면-간 높이이며, 납작머리 육각형은 높이보다 폭이 커서(cell_render_size가 tile_size·2/√3 × tile_size 반환) 더 넓은 스프라이트로 그려집니다. z는 직교처럼 고정 -1.0입니다. (예제 hex_tilemap_flat)

let atlas = TilemapAtlas::new("examples/assets/hex_tiles_flat.png", 2, 1);
// tile_size = 육각형 flat-to-flat 높이 (flat-top)
let tilemap = Tilemap::new(atlas, tiles, 64.0, Vec2::new(140.0, 110.0))
    .with_projection(TilemapProjection::HexagonalFlat);

오토타일은 투영과 무관 (격자에서 마스크 계산)

오토타일 비트마스크는 화면 배치가 아니라 tiles[row][col] 토폴로지에서 계산됩니다. 따라서 정사각용 Neighborhood::Edge4(이웃 4) / Blob8(이웃 8)은 직교와 아이소메트릭 모두에서 그대로 동작합니다 — 아이소메트릭은 동일한 정사각 격자를 다이아몬드로 그릴 뿐입니다. 육각 격자는 이웃 오프셋이 행/열 패리티에 따라 달라지므로 전용 6-이웃 이웃군이 필요합니다.

Blob8직교 / 아이소메트릭8 (감축 47개 마스크)N=1, E=2, S=4, W=8, NE=16, SE=32, SW=64, NW=128

아래는 2-타일 "내부/모서리" 규칙 예: 6개 이웃이 모두 채워진 헥스(마스크 63)만 잔디, 나머지(열린 변이 있는 헥스)는 모래를 그립니다. oob_filled: false이면 맵 가장자리 헥스는 모서리(모래)로 읽힙니다. (예제 hex_autotile)

use engine::{AutotileMode, Neighborhood, Tilemap, TilemapAutotile, TilemapProjection};
use std::collections::HashMap;

// 마스크 63 (6개 이웃 모두 채움)만 타일 0(잔디), 그 외는 타일 1(모래)
let mask_to_tile: HashMap<u8, u32> = (0u8..64)
    .map(|m| (m, if m == 63 { 0 } else { 1 }))
    .collect();
let autotile = TilemapAutotile {
    neighborhood: Neighborhood::Hex6,
    oob_filled: false, // 맵 가장자리 헥스 = 모서리(모래)
    mode: AutotileMode::Single { mask_to_tile },
};

let tilemap = Tilemap::new(atlas, tiles, 72.0, origin)
    .with_projection(TilemapProjection::Hexagonal);
// 같은 엔티티에 Tilemap + TilemapAutotile 둘 다 부착
world.add_component(map, tilemap);
world.add_component(map, autotile);

64-타일 헥사 blob 아틀라스 (타일 인덱스 == 6비트 마스크)

정사각 blob_47의 헥사 대응으로, 모든 열린 변을 올바르게 외곽선 처리하는 64-타일 blob 타일셋입니다. 연속된 64-타일 아틀라스에서 타일 인덱스가 곧 6비트 이웃 마스크가 되며, TilemapAutotile::hex_6(base_atlas_id)가 mask → base_atlas_id + mask 항등 매핑을 만들어 줍니다(납작머리는 hex_6_flat, Hex6Flat 비트 순서 사용). 두 생성자 모두 oob_filled는 기본 false이고, .with_oob_filled(true)로 외곽 경계를 벽(채움)으로 읽게 바꿀 수 있습니다. 64-타일 아틀라스는 cargo run --example gen_hex_autotile_sheet로 생성합니다. (예제 hex_blob_autotile)

// 8×8 = 64타일 뾰족머리 blob 시트: 타일 인덱스 == Hex6 이웃 마스크
let atlas = TilemapAtlas::new("examples/assets/hex_autotile.png", 8, 8);

// 항등 매핑(타일 0 + mask) — blob 시트의 핵심
let autotile = TilemapAutotile::hex_6(0);
//                                       ↑ base_atlas_id
// 납작머리라면: TilemapAutotile::hex_6_flat(0)  (Neighborhood::Hex6Flat)

let tilemap = Tilemap::new(atlas, carved_field, 72.0, origin)
    .with_projection(TilemapProjection::Hexagonal);
world.add_component(map, tilemap);
world.add_component(map, autotile);

구멍을 파거나 메우면 TilemapSystem이 영향받은 헥스의 마스크만 반응적으로 다시 계산하므로, 가장자리와 모든 구멍이 항상 올바른 외곽선을 유지합니다. 저수준에서 직접 마스크를 얻으려면 compute_tile_mask(tiles, row, col, nb, oob_filled)(또는 같은-지형 연결만 세는 compute_tile_mask_typed)를 사용합니다.

UI 포커스 내비게이션 (키보드 + 게임패드)

UiSystem의 포커스 패스는 UI 위젯에 키보드/게임패드 포커스를 자동으로 관리합니다. 게임은 위젯을 스폰하기만 하면 되고, 포커스 순환·활성화·포커스 링 그리기는 모두 엔진이 소유합니다. 포커스 가능한 위젯은 Button, TextInput, Slider, CheckBox이며, 포커스 순서는 위젯의 엔티티(스폰) 순서를 따릅니다. 숨김(hidden)/비활성(disabled) 위젯은 건너뜁니다.

UiFocus — 현재 포커스된 위젯

UiFocus는 현재 키보드 포커스를 가진 위젯을 담는 World 리소스입니다. 코어 리소스로 자동 삽입되므로 직접 삽입할 필요 없이 읽어서 무엇이 포커스되었는지 알 수 있습니다. 필드는 entity: Option<Entity> 하나이며 None은 포커스 없음을 의미합니다. 편의 메서드 is_focused(entity)로 특정 엔티티가 포커스 상태인지 확인합니다.

// 예제 ui_focus: 현재 포커스된 위젯 읽기 (엔진이 값을 갱신).
let focused = world
    .resource::<UiFocus>()
    .and_then(|f| f.entity)
    .map(|e| format!("entity {}", e.index()))
    .unwrap_or_else(|| "none".to_string());

포커스 패스는 매 프레임 입력을 읽어 다음을 처리합니다. 키보드 입력과 첫 번째로 연결된 게임패드의 입력이 하나로 합쳐지며, 게임패드/GamepadState가 없으면 게임패드 부분은 no-op입니다.

왼쪽 스틱 — 한 번 밀면 한 칸 (히스테리시스)

디지털 버튼(D-pad)은 엣지가 자동 감지되지만, 아날로그 스틱은 그렇지 않으므로 StickNav 엣지 검출기가 스틱을 이산적인 D-pad처럼 다룹니다. 스틱을 0.6(activate)을 넘겨 밀면 그 방향이 한 번만 발동하고, 다시 발동하려면 0.35(release) 중립 영역으로 돌아와야 합니다. 두 임계값 사이의 간격이 히스테리시스로, 누르고 있거나 떨리는 스틱이 자동 반복(auto-repeat)되는 것을 막습니다. 따라서 한 번 밀면 정확히 한 칸 이동합니다. (중립을 거치지 않고 반대 방향으로 곧장 꺾으면 새 방향이 발동합니다.)

WebAudio (wasm 오디오)

WebAudio는 네이티브 전용 AudioManager(rodio 기반)의 wasm 대응물입니다. 브라우저의 Web Audio API(AudioContext)를 얇게 감싼 래퍼로, 게임이 흔히 필요로 하는 기능 — 일회성 효과음, 스테레오 팬·볼륨·정지를 제어할 수 있는 SFX, 단일 루프 음악 채널(트랙 간 크로스페이드 포함), 마스터 볼륨, 이름 있는 믹서 버스(수동 더킹 포함), 2D 위치 기반 재생, 일시정지/재개 — 를 모두 제공합니다. 모든 재생 경로의 그래프는 노드 그래프(source → ... → master)로 구성되므로, 버스 볼륨·더킹·크로스페이드 같은 램프는 오디오 클럭(AudioParam) 위에서 스케줄됩니다. 즉 네이티브 AudioManager와 달리 매 프레임 update() 호출이 필요 없습니다.

WebAudio는 #[cfg(target_arch = "wasm32")]로 게이트되어 wasm 빌드에만 존재합니다. 브라우저는 사용자 제스처가 있기 전까지 AudioContext를 suspended 상태로 시작하므로, 첫 재생은 입력 핸들러에서 시작하거나 먼저 resume()을 호출해야 소리가 납니다. 보통 World 리소스로 저장해 시스템에서 구동합니다. 전체 표면은 예제 web_audio가 순서대로 실행하며 단계별 pass/fail을 기록하고, scripts/wasm_audio_smoke.sh가 헤드리스 브라우저로 오디오 그래프 수명주기를 자동 검증합니다(소리 자체는 직접 들어 확인).

기본 재생 + 마스터 볼륨

WebAudio::new()는 AudioContext와 마스터 게인을 만들고, 브라우저가 거부하면 None을 반환합니다. set_volume은 0.0..=1.0으로 클램프되며 모든 현재·미래 재생에 영향을 줍니다. play는 디코딩 후 한 번 재생하는 fire-and-forget 효과음입니다.

#[cfg(target_arch = "wasm32")]
{
    let Some(audio) = WebAudio::new() else { return };
    audio.set_volume(0.8);          // 마스터 볼륨 (0.0..=1.0 클램프)
    audio.resume();                 // 사용자 제스처 게이트 해제

    audio.play(jump_wav);           // fire-and-forget 일회성 효과음
    audio.play_music(theme_ogg);    // 단일 채널 루프 음악
    audio.crossfade_music(boss_ogg, 2.0); // 2초에 걸쳐 다른 트랙으로 크로스페이드
    // audio.stop_music();
    // audio.suspend(); audio.resume();   // 전체 일시정지 / 재개

    // 상태 조회 (예: "탭하여 소리 켜기" 프롬프트, 음악 on/off 토글)
    let _ = audio.is_running();
    let _ = audio.is_music_playing();
}

제어 가능한 SFX (Sfx 핸들) + 2D 위치 기반

play_sfx는 Sfx 핸들을 반환합니다. 핸들로 소스별 볼륨·스테레오 팬을 조절하거나 정지할 수 있습니다. 팬·볼륨 노드는 동기적으로 생성되므로 set_pan/set_volume은 디코딩이 끝나기 전에도 즉시 적용됩니다. play_at은 2D 위치 기반 재생으로, 거리에 따라 볼륨이 선형 감쇠(max_dist에서 무음)하고 x 오프셋에 따라 팬이 결정됩니다 — 움직이는 음원은 매 프레임 Sfx::update_position으로 추적합니다.

#[cfg(target_arch = "wasm32")]
{
    let sfx = audio.play_sfx(hit_wav);
    sfx.set_pan(-0.8);   // -1.0 왼쪽 .. 0.0 중앙 .. 1.0 오른쪽
    sfx.set_volume(0.5); // 마스터와 독립적인 소스별 볼륨
    // sfx.stop();

    // 2D 위치 기반: source/listener 좌표로 볼륨+팬 자동 계산
    let listener = Vec2::ZERO;
    let spatial = audio.play_at(step_wav, Vec2::new(50.0, 0.0), listener, 100.0);
    spatial.update_position(Vec2::new(-50.0, 0.0), listener, 100.0); // 매 프레임 갱신
}

이름 있는 믹서 버스 + 수동 더킹

버스는 사운드와 마스터 게인 사이의 duck → volume → master 게인 체인으로, 네이티브 믹서의 wasm 대응물입니다. 이름으로 사운드를 라우팅한 뒤 그룹 단위로 제어합니다. set_bus_volume(볼륨 노드)과 duck_bus/release_bus(더킹 노드)는 독립적으로 동작하므로 더킹 램프가 버스 볼륨을 덮어쓰지 않습니다. 버스는 첫 참조 시 지연 생성되며 — 볼륨만 설정한 버스도 등록되어 bus_names에 나타납니다.

주의 (유일한 비대칭): 자동 사이드체인 더킹(set_sidechain)은 트리거 버스가 재생 중인지 지속적으로 평가해야 하므로 fire-and-forget인 Web Audio 모델과 맞지 않아 네이티브 전용으로 남아 있습니다. wasm에서는 게임 로직에서 duck_bus/release_bus로 직접 더킹하세요.

#[cfg(target_arch = "wasm32")]
{
    audio.set_bus_volume("sfx", 0.5);     // 그룹 볼륨 (0.0..=1.0)
    audio.play_sfx_on_bus(hit_wav, "sfx"); // 이 SFX는 "sfx" 버스 볼륨을 따른다
    audio.play_on_bus(ui_wav, "ui");       // fire-and-forget도 버스를 등록

    // 수동 더킹: 음성/중요 큐가 재생되는 동안 음악·SFX를 낮춤
    audio.duck_bus("sfx", 0.2, 0.4);  // 0.4초에 걸쳐 더킹 (오디오 클럭 램프)
    audio.release_bus("sfx", 0.4);    // 다시 1.0으로 복귀

    let _ = audio.bus_volume("sfx");
    let _ = audio.bus_duck("sfx");

TimeScale / RealDt (시간 스케일)

TimeScale은 씬(scene) 시스템이 받는 dt(프레임 델타 시간)에 곱해지는 전역 배율 리소스다. 게임 전체의 시간 흐름을 한 줄로 조절할 수 있어 히트스톱, 슬로모, 패스트포워드 같은 게임 필 연출에 쓰인다. 1.0은 정상 속도, 0.0은 완전 정지(렌더링은 계속되는 히트스톱/일시정지), 0.3 같은 값은 슬로모, 2.0은 빨리 감기를 뜻한다.

중요한 점은 씬 시스템만 스케일링된다는 것이다. 엔진 내장 후처리(페이드, 핫리로드, 에셋 업로드, 카메라)와 꼬리(tail) 내장 시스템(계층 구조, 기즈모)은 항상 실시간(unscaled dt)으로 돌기 때문에, time_scale = 0.0으로 시간이 멈춘 동안에도 에디터와 화면 전환은 반응성을 유지한다. 음수 값은 TimeScale::get()에서 0.0으로 클램프된다.

// 히트스톱: 충돌 순간 시간을 거의 멈춘다 (juice_demo)
app.set_time_scale(0.06);   // 0.0 = 완전 정지, 0.3 = 슬로모, 1.0 = 정상, 2.0 = 빨리 감기

// 현재 배율 조회 (리소스가 없으면 1.0)
let scale = app.time_scale();

// 시스템 안에서 직접 조절할 수도 있다
if let Some(time_scale) = world.resource_mut::<TimeScale>() {
    time_scale.set(1.0); // 정상 속도로 복귀
}

씬 시스템의 run(&mut self, world, dt)에 전달되는 dt는 이미 TimeScale이 곱해진 값이다(scaled_dt = dt * time_scale.get()). 따라서 대부분의 게임플레이 로직은 평소처럼 인자로 받은 dt만 쓰면 자동으로 시간 스케일을 따른다.

RealDt — 시간 스케일을 무시하는 실제 dt

RealDt는 TimeScale이 적용되기 전의 실제(보정되지 않은) 프레임 델타 시간을 담은 리소스로, 매 프레임 갱신된다. 시간 스케일과 무관하게 실시간으로 돌아야 하는 시스템이 읽는다. 대표적인 예가 히트스톱 컨트롤러 자신이다. TimeScale(0.0)으로 세상을 멈추면, 그 시스템이 인자로 받는 dt도 0이 되어 버린다. 따라서 정지 시간(freeze window)을 보정된 dt로 세면 카운트다운이 영원히 멈춰 스스로를 풀 수 없게 된다. 이때 RealDt로 시간을 세면 멈춘 동안에도 타이머가 흐른다.

// juice_demo — 히트스톱 타이머는 RealDt(실제 dt)로 카운트다운한다
let real_dt = world.resource::<RealDt>().map(|r| r.0).unwrap_or(dt);

if impact {
    self.hitstop = 0.18;      // 0.18초 동안 정지
}
if self.hitstop > 0.0 {
    self.hitstop -= real_dt;  // ← scaled dt 대신 real_dt: 멈춘 중에도 줄어든다
}
let ts = if self.hitstop > 0.0 { 0.06 } else { 1.0 };
if let Some(time_scale) = world.resource_mut::<TimeScale>() {
    time_scale.set(ts);
}

juice_demo 예제(cargo run --example juice_demo, SPACE로 충돌)는 이 분리를 한눈에 보여 준다. 히트스톱 동안 게임플레이(스케일 dt로 도는 바운싱 스프라이트 행)는 얼어붙지만, 실시간으로 도는 카메라 셰이크와 비네트 펄스는 계속 애니메이션된다.

기타 신규 컴포넌트/시스템

이 절은 단독 심화 절을 두기에는 작지만 자주 쓰이는 6개 기능을 한데 모은 요약입니다. 명령형 타이밍 시퀀서(Coroutine), 연쇄 트윈(TweenSequence), 시차 스크롤(Parallax), 엔티티별 커스텀 셰이더(ShaderMaterial), 타일 애니메이션, 9-패치 스케일 스프라이트(NineSlice)를 다룹니다. 각 스니펫은 동봉된 예제에서 발췌한 실제로 컴파일되는 코드입니다.

Coroutine — 명령형 타이밍 시퀀서

Coroutine은 wait(secs) / run(|&mut World|) / run_for(dur, |&mut World, t|) 스텝을 빌더로 연결해 여러 프레임에 걸쳐 World에 임의의 클로저를 실행하는 시퀀서로, 웨이브 스폰·보스 인트로·컷신 로직 등 시나리오성 흐름에 적합합니다. CoroutineRunner 리소스에 start()로 등록하고 CoroutineSystem이 매 프레임 진행시킵니다(키프레임 기반 Timeline, 값 보간 TweenSequence와는 구분됨).

// 예제 coroutine_demo
let seq = Coroutine::new()
    .wait(1.0)                                   // 1초 대기
    .run(|world| { /* 박스 스폰 */ })            // 한 번 실행
    .run_for(2.0, |world, t| {                   // 2초 동안 매 프레임, t: 0→1
        if let Some(tr) = world.get_mut::<Transform>(e) {
            tr.position.x = start + (end - start) * t;
        }
    })
    .wait(0.5);

app.world.insert_resource(CoroutineRunner::new());
app.world.resource_mut::<CoroutineRunner>().unwrap().start(seq);
app.add_system(CoroutineSystem);
// 주의: 코루틴 클로저 안에서는 CoroutineRunner에 접근하지 말 것(틱 중에는 월드에서 제거됨)

TweenSequence — 연쇄 다단 트윈

TweenSequence는 then(start, end, duration, easing)으로 세그먼트를 이어 붙인 f32 트윈 체인입니다. 세그먼트마다 다른 이징을 줄 수 있고 looping(true)로 반복하며, tick(dt)가 현재 값을 반환하고 세그먼트 경계의 잔여 dt를 다음 세그먼트로 이월합니다. current_segment()·fraction()(전체 세그먼트 진행률)으로 상태를 조회합니다.

// 예제 tween_sequence
let mut seq_x = TweenSequence::new()
    .then(LEFT, RIGHT, LEG, Easing::EaseOut)
    .then(RIGHT, RIGHT, LEG, Easing::Linear)
    .then(RIGHT, LEFT, LEG, Easing::EaseInBack)
    .then(LEFT, LEFT, LEG, Easing::Linear)
    .looping(true);

let x = seq_x.tick(dt);          // 현재 값
let seg = seq_x.current_segment();
let frac = seq_x.fraction();

Parallax — 시차 스크롤 레이어

ParallaxLayer는 factor: Vec2 깊이 스크롤 계수를 갖는 컴포넌트입니다(1=월드 고정, 0=화면 고정, >1=전경). ParallaxSystem은 Camera를 읽어 pos = base + (cam - cam_ref) * (1 - factor)로 각 레이어 위치를 갱신하므로, 카메라를 움직이는 시스템 뒤에 추가합니다. 수평/수직 전용은 horizontal(fx)·vertical(fy)로 만듭니다.

// 예제 parallax_scroll
let e = world.spawn();
world.add_component(e, Transform { /* ... */ });
world.add_component(e, Sprite::colored(0.16, 0.20, 0.42));
world.add_component(e, ParallaxLayer::horizontal(0.10)); // 먼 하늘, 거의 안 움직임

app.add_system(CameraFollowSystem { /* ... */ });
app.add_system(ParallaxSystem);                          // 카메라 이동 시스템 뒤에

ShaderMaterial — 엔티티별 커스텀 프래그먼트 셰이더

ShaderMaterial은 엔티티에 붙여 내장 스프라이트 프래그먼트 셰이더를 대체하는 컴포넌트입니다. WGSL 바인딩은 @group(1) 텍스처/샘플러와 @group(2)의 params: vec4<f32>이며, 렌더러가 소스 해시별로 파이프라인을 1개씩 컴파일·캐시합니다. params를 매 프레임 갱신해 시간·임계값 등을 셰이더로 전달합니다.

// 예제 shader_material
let e = app.world.spawn();
app.world.add_component(e, Transform { /* ... */ });
app.world.add_component(e, Sprite::colored(1.0, 1.0, 1.0));
app.world.add_component(e, ShaderMaterial::new(SHADER_PLASMA, [0.0; 4]));

// 시스템에서 params 갱신 (params[0] = 경과 시간)
if let Some(mat) = world.get_mut::<ShaderMaterial>(e) {
    mat.params[0] = self.elapsed;
}

애니메이션 타일 — TileAnimationSet · AnimatedTileSystem

TileAnimation(프레임 인덱스 목록 + frame_time)을 TileAnimationSet에 타일 값별로 등록하고, 그 셋을 Tilemap 엔티티에 컴포넌트로 붙입니다. AnimatedTileSystem이 TilemapSystem 뒤에서 태그된 셀의 UvRect만 매 프레임 갱신하므로(렌더 전용, 제너레이션을 올리지 않음) 비-애니메이션 빠른 경로는 그대로 유지됩니다.

// 예제 animated_tiles
let mut anim_set = TileAnimationSet::new();
anim_set.insert(1, TileAnimation::new(vec![0, 1, 2, 3], 0.18)); // 물: 프레임 0~3
anim_set.insert(2, TileAnimation::new(vec![4, 5, 6], 0.22));    // 용암: 프레임 4~6
// 등록 안 한 값 → 정적 타일

let map_e = app.world.spawn();
app.world.add_component(map_e, tilemap);
app.world.add_component(map_e, anim_set);

app.add_system(TilemapSystem::new());     // 먼저: 타일 엔티티 생성 + 태그
app.add_system(AnimatedTileSystem::new()); // 뒤: 클럭 진행 + UvRect 갱신

NineSlice — 9-패치 스케일 스프라이트

NineSlice는 모서리 크기를 고정한 채 가장자리/중앙만 늘려 그리는 9-패치 컴포넌트입니다. border: [f32;4](월드 px)와 uv_border: [f32;4](UV 비율) 모두 [left, right, top, bottom] 순이며, 균일한 테두리는 uniform(border_px, uv_frac)로 만듭니다. 스프라이트 패스가 NineSlice가 있을 때만 9개 서브-쿼드를 발행하므로(가산적, 일반 스프라이트는 변화 없음) 모서리가 늘어지지 않고 강체로 회전합니다(AtlasSprite/ShaderMaterial에는 사용 불가).

// 예제 nine_slice
let e = app.world.spawn();
app.world.add_component(e, Transform {
    position: Vec2::new(cx, cy),
    scale: Vec2::new(w, h),         // 어떤 크기로 늘려도 모서리는 고정
    ..Default::default()
});
app.world.add_component(e, Sprite::with_handle(handle.clone()));
// 16px 모서리, 64px 텍스처이므로 UV 테두리 = 16/64 = 0.25
app.world.add_component(e, NineSlice::uniform(16.0, 16.0 / 64.0));

v10.5.0 — Coroutine 시퀀서

대기 / 일회성 실행 / 구간 실행을 순서대로 엮는 명령형 스크립트. 컷씬·웨이브 스폰·보스 인트로 같은 시간 기반 게임 로직에 적합하다. Timeline(데이터 키프레임), TweenSequence(값 보간)와 구별된다.

use engine::{Coroutine, CoroutineRunner, CoroutineSystem};

let seq = Coroutine::new()
    .wait(1.0)
    .run(|world| { /* 한 번 실행: 스폰 등 */ })
    .run_for(2.0, |world, t| { /* 매 프레임, t는 0→1 */ })
    .wait(0.5)
    .run(|world| { /* 보스 스폰 */ });

app.world.insert_resource(CoroutineRunner::new());
app.world.resource_mut::<CoroutineRunner>().unwrap().start(seq);
app.add_system(CoroutineSystem);
borrow 안전: CoroutineSystem은 틱 동안 CoroutineRunner를 월드에서 잠깐 빼낸다(remove→tick→reinsert). 덕분에 클로저가 &mut World를 자유롭게 받는다. 클로저 안에서 CoroutineRunner에 재진입하지 말 것.
